# KTA Web Admin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Next.js/React web admin interface for managing and previewing KTA cards using the existing Paket E-Card SVG component.

**Architecture:** Reuse the existing `KtaCard` SVG component directly. Create admin pages for member lookup, KTA preview (both sides), validation warnings, PNG export, and print-ready view. Integrate with existing authentication and API layer.

**Tech Stack:** Next.js 14+ (App Router), React 18+, TypeScript, `qrcode` library, existing `KtaCard` component from Paket E-Card.

**Spec:** `Paket E-Card/.stitch/DESIGN.md`

## Global Constraints

- Use existing `KtaCard` component as-is — no visual modifications
- Card dimensions: CR80 (85.6 × 53.98 mm trim, 2mm bleed)
- Font: DejaVu Sans (regular/bold) via `kta-fonts.css` imported at root
- Dependencies: `qrcode`, `@types/qrcode`
- Asset paths: `/kta/background/depan.png`, `/kta/background/belakang.png`, `/kta/logo.png`, `/kta/fonts/DejaVuSans*.ttf`
- Data model: `Anggota` interface from `types.ts`
- PNG export: 2022 × 1275 px (≈600 DPI) with embedded fonts/images
- Admin routes must be protected by authentication
- QR verification URLs must be generated by backend, not frontend
- Indonesian locale: `toLocaleUpperCase('id-ID')` for all display text

## Review Focus

1. **Missing/invalid member data** → Validation warnings display, export disabled, no silent truncation
2. **CORS-blocked images** (photo, QR, signature) → Export fails with clear error, preview shows placeholder
3. **QR URL too long for QR code** → Warning shown, fallback to manual entry or custom QR image
4. **Unauthorized admin access** → Redirect to login, no data leakage
5. **Print layout mismatch** → `@page` CSS produces correct physical dimensions with bleed

```tsx
// components/kta/KtaAdminPreview.tsx
'use client';
import { useRef, useState, useCallback } from 'react';
import { KtaCard } from './KtaCard';
import { downloadCardPng } from './exportPng';
import type { Anggota } from './types';

interface Props {
  anggota: Anggota;
}

export function KtaAdminPreview({ anggota }: Props) {
  const depanRef = useRef<SVGSVGElement>(null);
  const belakangRef = useRef<SVGSVGElement>(null);
  const [warnings, setWarnings] = useState<string[]>([]);
  const [exportError, setExportError] = useState('');

  const handleWarnings = useCallback((msgs: string[]) => setWarnings(msgs), []);

  const unduh = async (side: 'depan' | 'belakang') => {
    const svg = side === 'depan' ? depanRef.current : belakangRef.current;
    if (!svg || warnings.length) return;
    try {
      setExportError('');
      await downloadCardPng(svg, side);
    } catch (e) {
      setExportError(e instanceof Error ? e.message : 'Ekspor gagal');
    }
  };

  const cetak = () => window.print();

  return (
    <section className="kta-admin-preview">
      {warnings.length > 0 && (
        <ul role="alert" className="warnings">
          {warnings.map(w => <li key={w}>{w}</li>)}
        </ul>
      )}
      {exportError && <p role="alert" className="error">{exportError}</p>}

      <div className="card-preview">
        <h3>Depan</h3>
        <KtaCard side="depan" anggota={anggota} svgRef={depanRef} onWarnings={handleWarnings} />
      </div>
      <div className="card-preview">
        <h3>Belakang</h3>
        <KtaCard side="belakang" anggota={anggota} svgRef={belakangRef} />
      </div>

      <div className="actions">
        <button disabled={warnings.length > 0} onClick={() => unduh('depan')}>
          Unduh Depan (PNG)
        </button>
        <button disabled={warnings.length > 0} onClick={() => unduh('belakang')}>
          Unduh Belakang (PNG)
        </button>
        <button onClick={cetak}>Cetak (Ctrl+P)</button>
      </div>
    </section>
  );
}
```

```tsx
// components/kta/AnggotaSearch.tsx
'use client';
import { useState, useTransition } from 'react';
import Link from 'next/link';
import { searchAnggota } from '@/lib/kta/api';
import type { Anggota } from '@/components/kta/types';

export function AnggotaSearch() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<Anggota[]>([]);
  const [isPending, startTransition] = useTransition();

  const handleSearch = async (q: string) => {
    setQuery(q);
    if (q.length < 2) { setResults([]); return; }
    startTransition(async () => {
      try {
        const data = await searchAnggota(q);
        setResults(data);
      } catch { setResults([]); }
    });
  };

  return (
    <section>
      <input
        type="search"
        placeholder="Cari nama atau nomor anggota..."
        value={query}
        onChange={e => handleSearch(e.target.value)}
        className="search-input"
      />
      {isPending && <span>Mencari...</span>}
      <ul className="results">
        {results.map(a => (
          <li key={a.nomor}>
            <Link href={`/admin/kta/${a.nomor}`}>
              <strong>{a.nama}</strong> — {a.nomor} ({a.ranting})
            </Link>
          </li>
        ))}
      </ul>
    </section>
  );
}
```
